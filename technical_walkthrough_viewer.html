import { createSlice, PayloadAction } from "@reduxjs/toolkit"
import { RCARecord, TrackedField, WhyStep } from "../types"

export interface InvestigationState {
  id: string | null
  record: RCARecord
  dirty: boolean
  highlighted: string[]  // Visual cues for fresh AI extractions
  lastSavedAt: string | null
}

export const investigationSlice = createSlice({
  name: "investigation",
  initialState,
  reducers: {
    applyServerRecord(state, action: PayloadAction<{ record: RCARecord; changed: string[] }>) {
      state.record = action.payload.record
      state.highlighted = action.payload.changed
      state.dirty = true
    },
    confirmRootCause(state, action: PayloadAction<{ id: string; investigator: string }>) {
      const cause = state.record.root_causes.find(c => c.id === action.payload.id)
      if (cause) {
        cause.confirmed = true
        cause.confirmed_by = action.payload.investigator
      }
    }
  }
})

def evidence_grounded(evidence: str, source: str) -> bool:
    quote = " ".join((evidence or "").split())
    source_folded = source.lower()
    
    # 1. Direct exact substring match
    if quote.lower() in source_folded:
        return True
        
    # 2. Strict numerical verification (prevents hallucinating numbers)
    numbers = _NUMBER.findall(quote)
    if any(number not in source_folded for number in numbers):
        return False
        
    # 3. Content token lexical overlap (>= 45% threshold)
    tokens = [t for t in _TOKEN.findall(quote.lower()) if len(t) >= 4 and t not in _STOP]
    source_tokens = set(_TOKEN.findall(source_folded))
    overlap = sum(1 for t in tokens if t in source_tokens)
    return overlap >= 3 and (overlap / len(tokens)) >= 0.45
Anti-Hallucination Guardrails
  • Zero Synthetic Numbers Rejects quotes containing unverified percentages, temperatures, or batch IDs.
  • Explicit Gap Output Missing info produces explicit Evidence Gaps rather than fabricated facts.
  • Human Sign-Off Boundary AI generates hypotheses; only qualified investigators confirm findings.
